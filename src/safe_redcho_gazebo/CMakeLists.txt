cmake_minimum_required(VERSION 3.16)
project(safe_redcho_gazebo)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(safe_redcho_motor_system SHARED src/motor_thrust_system.cpp)
target_compile_features(safe_redcho_motor_system PUBLIC cxx_std_17)
target_include_directories(safe_redcho_motor_system PRIVATE include)
ament_target_dependencies(safe_redcho_motor_system rclcpp nav_msgs std_msgs)
target_link_libraries(safe_redcho_motor_system gz-sim8::gz-sim8 gz-plugin2::register)
install(TARGETS safe_redcho_motor_system LIBRARY DESTINATION lib)
ament_package()
