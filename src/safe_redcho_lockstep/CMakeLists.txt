cmake_minimum_required(VERSION 3.16)
project(safe_redcho_lockstep)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(safe_redcho_interfaces REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz_transport_vendor REQUIRED)
find_package(gz_msgs_vendor REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(Threads REQUIRED)
add_library(safe_redcho_lockstep_system SHARED src/lockstep_system.cpp)
target_compile_features(safe_redcho_lockstep_system PUBLIC cxx_std_17)
target_include_directories(safe_redcho_lockstep_system PRIVATE include)
ament_target_dependencies(safe_redcho_lockstep_system
  rclcpp nav_msgs std_msgs safe_redcho_interfaces)
target_link_libraries(safe_redcho_lockstep_system
  gz-sim8::gz-sim8 gz-plugin2::register Threads::Threads
  gz-transport13::gz-transport13 gz-msgs10::gz-msgs10)
install(TARGETS safe_redcho_lockstep_system LIBRARY DESTINATION lib)
ament_package()
